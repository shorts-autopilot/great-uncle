# Phase 4 demo: undo, move-in-black and timecode.
#
#   cargo run -p openlight-console -- --output 1=virtual \
#     --script examples/phase4-demo.txt --frames 120 --trace
#
# The script is deterministic: the track has been stopped, and the final frame
# is half a second into sequence 3's per-attribute cue.

Import Gdtf "examples/gdtf/openlight-demo-wash.gdtf"

Patch Fixture 1 At 1.001 Type "OpenLight Demo Wash"
Patch Fixture 2 At 1.005 Type "OpenLight Demo Wash"

# Undo and redo cover show edits. The programmer keeps its value, because it is
# console state, not show data.
Fixture 1 Thru 2 At 40
Master Grand At 50
Undo
Redo
Undo

# Cue 1 is black; cue 2 brings the dimmer up with move in black, so the colour
# is already in place before the fixtures come on.
Fixture 1 Thru 2 At 0
Fixture 1 Thru 2 At ColorAdd_R 0
Store Cue 1 "Dark" Fade 0
Fixture 1 Thru 2 At Full
Fixture 1 Thru 2 At ColorAdd_R Full
Store Cue 2 "Red" Fade 1 Mib

# A timecode show fires the two cues from the engine clock.
Clear All
Store Timecode 1 "Song" Fps 25
Timecode 1 Add 00:00:00:00 Sequence 1 Cue 1
Timecode 1 Add 00:00:01:00 Sequence 1 Cue 2
Timecode 1 Track 2 "Cues"
Timecode 1 Track 2 Add 00:00:00:12 Sequence 1 Cue 1
Timecode 1 Marker 00:00:01:00 "Red"
Timecode 1 Start
Wait 2
Timecode 1 Pause
Timecode 1 Stop
# The track leaves cue 2 holding the rig; release it so the timing scene below
# is the only playback in the final frame.
Release Sequence 1

# A list that runs itself: cue 2 follows cue 1's fade with no command in
# between. It is released afterwards so the final frame is unchanged.
Fixture 1 Thru 2 At Full
Store Cue 1 Sequence 2 Fade 0
Fixture 1 Thru 2 At 0
Store Cue 2 Sequence 2 Follow 0.5 Fade 1
Go Sequence 2
Wait 1.5
Release Sequence 2
Clear All

# Per-attribute timing: the colour is in place after half a second while the
# dimmer waits six seconds and then snaps, which is past the end of this run.
# The final frame shows the colour full and the dimmer still dark, so the trace
# proves the order.
Fixture 1 Thru 2 At 0
Fixture 1 Thru 2 At ColorAdd_R 0
Store Cue 1 "Dark" Fade 0 Sequence 3
Fixture 1 Thru 2 At ColorAdd_R Full
Fixture 1 Thru 2 At Full
Store Cue 2 "Colour first" Fade 3 Sequence 3
Cue 2 Timing ColorAdd_R Fade 0.5 Sequence 3
Cue 2 Timing Dimmer Delay 6 Curve Snap Sequence 3
Clear All
Go Sequence 3
Go Sequence 3
# Stop half a second in: the colour has arrived and the dimmer has not.
Wait 0.5

Timecode 1
Sequence 1
Sequence 2
Sequence 3
Status
Save "target/phase4-demo.olshow.json"
